# 生产网络抓包常用 TShark / Wireshark Display Filter 表达式精选

# 1. TCP 层异常诊断
tcp.analysis.retransmission          # 抓出所有 TCP 重发包
tcp.analysis.fast_retransmission     # 抓出 TCP 快速重传包
tcp.analysis.duplicate_ack           # 提取重复 ACK 确认包 (拥塞或丢包先兆)
tcp.analysis.zero_window             # 接收方缓冲区已满，宣布零窗口
tcp.flags.reset == 1                 # 抓取 TCP 连接被主动 RST 重置的会话

# 2. HTTP 协议层精准排障
http.response.code >= 500            # 服务端内部错误 (500/502/503/504)
http.time > 1.0                      # 请求响应耗时超过 1 秒的慢请求
http.request.method == "POST"        # 过滤所有 POST 请求
http.user_agent contains "Baidu"     # 提取百度蜘蛛 UA 流量

# 3. DNS 与 TLS 握手层排查
dns.flags.response == 1 && dns.time > 0.1 # DNS 解析耗时超过 100ms
tls.handshake.type == 1                   # 提取 TLS Client Hello
tls.alert_message.level == 2              # 提取 TLS 握手 Fatal 致命错误
